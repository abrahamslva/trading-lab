    def calculate_keltner(self, df, period=20):
        """Calcular Keltner Channels"""
        ema = talib.EMA(df['close'], timeperiod=period)
        atr = talib.ATR(df['high'], df['low'], df['close'], timeperiod=period)
        upper = ema + (2 * atr)
        lower = ema - (2 * atr)
        return ema, upper, lower
    
    def calculate_vwap(self, df):
        """Calcular VWAP"""
        tp = (df['high'] + df['low'] + df['close']) / 3
        vwap = (tp * df['volume']).cumsum() / df['volume'].cumsum()
        return vwap
    
    def calculate_cmf(self, df):
        """Calcular Chaikin Money Flow"""
        mfv = ((df['close'] - df['low']) - (df['high'] - df['close'])) / (df['high'] - df['low']) * df['volume']
        cmf = mfv.rolling(window=20).sum() / df['volume'].rolling(window=20).sum()
        return cmf
    
    def calculate_support_resistance(self, df, window=20):
        """Calcular niveles de soporte y resistencia"""
        highs = df['high'].rolling(window=window).max()
        lows = df['low'].rolling(window=window).min()
        return lows, highs
    
    def calculate_trend_strength(self, df):
        """Calcular fuerza de tendencia"""
        sma_20 = talib.SMA(df['close'], timeperiod=20)
        sma_50 = talib.SMA(df['close'], timeperiod=50)
        trend_strength = abs(sma_20 - sma_50) / sma_50
        return trend_strength
    
    def calculate_volatility_regime(self, indicators):
        """Calcular régimen de volatilidad"""
        atr = indicators['atr']
        atr_ma = atr.rolling(window=50).mean()
        volatility_ratio = atr / atr_ma
        return volatility_ratio

# ==================== SISTEMA DE PONDERACIÓN DINÁMICA ====================
class DynamicWeightingSystem:
    def __init__(self):
        self.market_regime = 'unknown'
        self.volatility_state = 'normal'
        self.performance_history = {}
    
    def update_market_regime(self, indicators):
        """Actualizar régimen del mercado"""
        latest = {k: v.iloc[-1] if isinstance(v, pd.Series) else v for k, v in indicators.items()}
        
        # Determinar régimen de tendencia
        adx = latest.get('adx', 0)
        trend_strength = latest.get('trend_strength', 0)
        
        if adx > 30 and trend_strength > 0.02:
            self.market_regime = 'trending'
        elif adx < 20 and trend_strength < 0.01:
            self.market_regime = 'ranging'
        else:
            self.market_regime = 'transition'
        
        # Determinar estado de volatilidad
        volatility_regime = latest.get('volatility_regime', 1)
        if volatility_regime > 1.5:
            self.volatility_state = 'high'
        elif volatility_regime < 0.7:
            self.volatility_state = 'low'
        else:
            self.volatility_state = 'normal'
    
    def calculate_indicator_weights(self, strategy, indicators):
        """Calcular pesos dinámicos de indicadores según condiciones del mercado"""
        base_weights = GlobalConfig.INDICATOR_WEIGHTS[strategy].copy()
        
        # Ajustar pesos según régimen del mercado
        if self.market_regime == 'trending':
            if strategy == 'trend':
                base_weights['adx'] *= 1.3
                base_weights['macd'] *= 1.2
                base_weights['ema'] *= 1.2
            elif strategy == 'mean_reversion':
                base_weights['bollinger'] *= 0.7
                base_weights['rsi'] *= 0.8
        
        elif self.market_regime == 'ranging':
            if strategy == 'mean_reversion':
                base_weights['bollinger'] *= 1.3
                base_weights['rsi'] *= 1.2
                base_weights['stochastic'] *= 1.2
            elif strategy == 'trend':
                base_weights['adx'] *= 0.7
                base_weights['parabolic'] *= 0.8
        
        # Ajustar según volatilidad
        if self.volatility_state == 'high':
            if strategy == 'breakout':
                base_weights['volume'] *= 1.3
                base_weights['atr'] *= 1.2
            else:
                base_weights['rsi'] *= 0.8
                base_weights['stochastic'] *= 0.8
        
        # Normalizar pesos
        total_weight = sum(base_weights.values())
        normalized_weights = {k: v/total_weight for k, v in base_weights.items()}
        
        return normalized_weights
    
    def calculate_strategy_score(self, strategy, indicators, signal):
        """Calcular puntuación de la estrategia con pesos dinámicos"""
        weights = self.calculate_indicator_weights(strategy, indicators)
        latest = {k: v.iloc[-1] if isinstance(v, pd.Series) else v for k, v in indicators.items()}
        
        score = 0.0
        
        if strategy == 'harmonic':
            score += self._calculate_harmonic_score(latest, weights, signal)
        elif strategy == 'trend':
            score += self._calculate_trend_score(latest, weights)
        elif strategy == 'mean_reversion':
            score += self._calculate_mean_reversion_score(latest, weights)
        elif strategy == 'momentum':
            score += self._calculate_momentum_score(latest, weights)
        elif strategy == 'breakout':
            score += self._calculate_breakout_score(latest, weights)
        elif strategy == 'ml_prediction':
            score += self._calculate_ml_score(latest, weights, signal)
        
        return min(score, 1.0)
    
    def _calculate_harmonic_score(self, latest, weights, signal):
        """Calcular puntuación para estrategia armónica"""
        score = signal.get('confidence', 0) * weights.get('fibonacci', 0)
        
        # Confirmaciones técnicas
        if 30 < latest.get('rsi', 50) < 70:
            score += weights.get('rsi', 0) * 0.8
        
        if latest.get('macd_hist', 0) > 0:
            score += weights.get('macd', 0) * 0.7
        
        if latest.get('volume_ratio', 1) > 1.2:
            score += weights.get('volume', 0) * 0.6
        
        return score
    
    def _calculate_trend_score(self, latest, weights):
        """Calcular puntuación para estrategia de tendencia"""
        score = 0.0
        
        # Fuerza de tendencia
        if latest.get('adx', 0) > 25:
            score += weights.get('adx', 0) * min(latest['adx'] / 50, 1.0)
        
        # Momentum
        if latest.get('macd_hist', 0) > 0:
            score += weights.get('macd', 0) * 0.8
        
        # Media móvil
        if latest.get('ema_12', 0) > latest.get('ema_26', 0):
            score += weights.get('ema', 0) * 0.7
        
        # SAR
        if latest.get('parabolic_sar', 0) < latest.get('close', 0):
            score += weights.get('parabolic', 0) * 0.6
        
        # Volumen
        if latest.get('volume_ratio', 1) > 1.1:
            score += weights.get('volume', 0) * 0.5
        
        return score
    
    def _calculate_mean_reversion_score(self, latest, weights):
        """Calcular puntuación para mean reversion"""
        score = 0.0
        
        # Bollinger Bands
        bb_position = (latest.get('close', 0) - latest.get('bb_lower', 0)) / \
                     (latest.get('bb_upper', 0) - latest.get('bb_lower', 0))
        
        if bb_position < 0.1 or bb_position > 0.9:
            score += weights.get('bollinger', 0) * 0.9
        
        # RSI extremo
        rsi = latest.get('rsi', 50)
        if rsi < 30 or rsi > 70:
            score += weights.get('rsi', 0) * 0.8
        
        # Stochastic extremo
        stoch_k = latest.get('stoch_k', 50)
        if stoch_k < 20 or stoch_k > 80:
            score += weights.get('stochastic', 0) * 0.7
        
        # CCI extremo
        cci = latest.get('cci', 0)
        if cci < -150 or cci > 150:
            score += weights.get('cci', 0) * 0.6
        
        # Williams %R extremo
        williams = latest.get('williams_r', -50)
        if williams < -80 or williams > -20:
            score += weights.get('williams', 0) * 0.5
        
        return score
    
    def _calculate_momentum_score(self, latest, weights):
        """Calcular puntuación para momentum"""
        score = 0.0
        
        # RSI momentum
        rsi = latest.get('rsi', 50)
        if 40 < rsi < 60:
            score += weights.get('rsi', 0) * 0.8
        
        # MACD momentum
        if latest.get('macd_hist', 0) > 0:
            score += weights.get('macd', 0) * 0.8
        
        # Stochastic momentum
        stoch_k = latest.get('stoch_k', 50)
        if 20 < stoch_k < 80:
            score += weights.get('stochastic', 0) * 0.7
        
        # Volumen confirmación
        if latest.get('volume_ratio', 1) > 1.3:
            score += weights.get('volume', 0) * 0.6
        
        # MFI momentum
        mfi = latest.get('mfi', 50)
        if 30 < mfi < 70:
            score += weights.get('mfi', 0) * 0.5
        
        # ROC momentum
        roc = latest.get('roc', 0)
        if abs(roc) > 2:
            score += weights.get('roc', 0) * 0.4
        
        return score
    
    def _calculate_breakout_score(self, latest, weights):
        """Calcular puntuación para breakout"""
        score = 0.0
        
        # Volumen fuerte
        volume_ratio = latest.get('volume_ratio', 1)
        if volume_ratio > 2.0:
            score += weights.get('volume', 0) * min(volume_ratio / 3, 1.0)
        
        # Breakout de Bollinger
        close = latest.get('close', 0)
        bb_upper = latest.get('bb_upper', 0)
        bb_lower = latest.get('bb_lower', 0)
        
        if close > bb_upper or close < bb_lower:
            score += weights.get('bollinger', 0) * 0.9
        
        # Fuerza de tendencia
        if latest.get('adx', 0) > 30:
            score += weights.get('adx', 0) * min(latest['adx'] / 50, 1.0)
        
        # Volatilidad alta
        atr = latest.get('atr', 0)
        if atr > latest.get('atr_7', 0) * 1.2:
            score += weights.get('atr', 0) * 0.7
        
        # Breakout de Donchian
        donchian_upper = latest.get('donchian_upper', 0)
        donchian_lower = latest.get('donchian_lower', 0)
        
        if close > donchian_upper or close < donchian_lower:
            score += weights.get('donchian', 0) * 0.6
        
        return score
    
    def _calculate_ml_score(self, latest, weights, signal):
        """Calcular puntuación para ML prediction"""
        score = 0.0
        
        # Puntuación del modelo ML
        ml_confidence = signal.get('confidence', 0)
        score += weights.get('ml_score', 0) * ml_confidence
        
        # Confirmación técnica
        tech_score = self._calculate_technical_confirmation(latest)
        score += weights.get('technical_score', 0) * tech_score
        
        # Confirmación de volumen
        volume_ratio = latest.get('volume_ratio', 1)
        if volume_ratio > 1.2:
            score += weights.get('volume', 0) * min(volume_ratio / 2, 1.0)
        
        return score
    
    def _calculate_technical_confirmation(self, latest):
        """Calcular confirmación técnica para ML"""
        confirmations = 0
        total_checks = 0
        
        # RSI confirmation
        rsi = latest.get('rsi', 50)
        if 30 < rsi < 70:
            confirmations += 1
        total_checks += 1
        
        # MACD confirmation
        if latest.get('macd_hist', 0) != 0:
            confirmations += 1
        total_checks += 1
        
        # Trend confirmation
        if latest.get('adx', 0) > 20:
            confirmations += 1
        total_checks += 1
        
        # Volume confirmation
        if latest.get('volume_ratio', 1) > 1.1:
            confirmations += 1
        total_checks += 1
        
        return confirmations / total_checks if total_checks > 0 else 0

        features['volume_change'] = df['volume'].pct_change()
        features['volume_sma_ratio'] = indicators['volume'] / indicators['volume_sma']
        features['volume_ema_ratio'] = indicators['volume'] / indicators['volume_ema']
        features['obv'] = indicators['obv']
        features['obv_change'] = features['obv'].pct_change()
        features['vwap'] = indicators['vwap']
        features['price_vs_vwap'] = (df['close'] - features['vwap']) / features['vwap']
        features['ad_line'] = indicators['ad_line']
        features['cmf'] = indicators['cmf']
        
        # Características de tendencia (expandido)
        features['adx'] = indicators['adx']
        features['adxr'] = indicators['adxr']
        features['aroon_up'] = indicators['aroon_up']
        features['aroon_down'] = indicators['aroon_down']
        features['aroon_osc'] = indicators['aroon_osc']
        features['trend_strength'] = indicators['trend_strength']
        features['parabolic_sar'] = indicators['parabolic_sar']
        features['price_vs_sar'] = (df['close'] - features['parabolic_sar']) / df['close']
        
        # Características de patrones
        features['doji'] = indicators['doji']
        features['hammer'] = indicators['hammer']
        features['engulfing'] = indicators['engulfing']
        features['morning_star'] = indicators['morning_star']
        features['evening_star'] = indicators['evening_star']
        
        # Características de estructura de mercado
        features['support_distance'] = (df['close'] - indicators['support']) / df['close']
        features['resistance_distance'] = (indicators['resistance'] - df['close']) / df['close']
        features['volatility_regime'] = indicators['volatility_regime']
        
        # Características temporales
        features['hour'] = df.index.hour
        features['day_of_week'] = df.index.dayofweek
        features['day_of_month'] = df.index.day
        features['month'] = df.index.month
        features['quarter'] = df.index.quarter
        
        # Características de interacción
        features['rsi_volume'] = features['rsi'] * features['volume_ratio']
        features['macd_volume'] = features['macd_hist'] * features['volume_ratio']
        features['adx_volume'] = features['adx'] * features['volume_ratio']
        features['volatility_trend'] = features['volatility_20'] * features['trend_strength']
        
        # Eliminar NaN
        features = features.dropna()
        
        self.feature_names = features.columns.tolist()
        return features
    
    def train_models(self, historical_data):
        """Entrenar modelos de ML con selección de características"""
        print("Entrenando modelos de ML mejorados...")
        
        all_features = []
        all_directions = []
        all_magnitudes = []
        
        for pair_data in historical_data:
            if len(pair_data) < 200:
                continue
                
            indicators = AdvancedIndicators().calculate_all_indicators(pair_data)
            features = self.prepare_enhanced_features(pair_data, indicators)
            directions, magnitudes = self.prepare_targets(pair_data)
            
            # Alinear características y targets
            min_length = min(len(features), len(directions), len(magnitudes))
            if min_length > 100:
                all_features.append(features.iloc[:min_length])
                all_directions.append(directions.iloc[:min_length])
                all_magnitudes.append(magnitudes.iloc[:min_length])
        
        if not all_features:
            print("No hay suficientes datos para entrenar")
            return
        
        # Combinar todos los datos
        X = pd.concat(all_features, ignore_index=True)
        y_direction = pd.concat(all_directions, ignore_index=True)
        y_magnitude = pd.concat(all_magnitudes, ignore_index=True)
        
        # Dividir datos
        X_train, X_test, y_train, y_test = train_test_split(X, y_direction, test_size=0.2, random_state=42)
        
        # Selección de características
        print("Seleccionando características más importantes...")
        selector = SelectKBest(score_func=f_classif, k=50)
        X_train_selected = selector.fit_transform(X_train, y_train)
        X_test_selected = selector.transform(X_test)
        
        # Entrenar cada modelo
        for model_name, model in self.models.items():
            print(f"Entrenando modelo {model_name}...")
            
            # Escalar características
            if model_name == 'mlp':
                scaler = RobustScaler()
                X_train_scaled = scaler.fit_transform(X_train_selected)
                X_test_scaled = scaler.transform(X_test_selected)
            else:
                scaler = StandardScaler()
                X_train_scaled = scaler.fit_transform(X_train_selected)
                X_test_scaled = scaler.transform(X_test_selected)
            
            # Entrenar modelo
            model.fit(X_train_scaled, y_train)
            
            # Guardar scaler y selector
            self.scalers[model_name] = scaler
            self.feature_selector[model_name] = selector
            
            # Evaluar
            accuracy = model.score(X_test_scaled, y_test)
            print(f"Modelo {model_name} - Accuracy: {accuracy:.3f}")
        
        self.is_trained = True
        print("Entrenamiento completado")
    
    def predict(self, df, indicators):
        """Realizar predicción con ensemble mejorado"""
        if not self.is_trained:
            return None
        
        features = self.prepare_enhanced_features(df, indicators)
        if len(features) == 0:
            return None
        
        # Obtener última fila de características
        latest_features = features.iloc[-1:].values
        
        # Predicciones de cada modelo
        predictions = {}
        for model_name, model in self.models.items():
            if model_name in self.scalers:
                selector = self.feature_selector[model_name]
                scaler = self.scalers[model_name]
                
                # Seleccionar y escalar características
                features_selected = selector.transform(latest_features)
                features_scaled = scaler.transform(features_selected)
                
                # Obtener predicción
                pred_proba = model.predict_proba(features_scaled)[0]
                predictions[model_name] = pred_proba
        
        # Ensemble ponderado
        ensemble_pred = np.zeros(2)
        for model_name, pred in predictions.items():
            weight = self.model_weights[model_name]
            ensemble_pred += pred * weight
        
        # Calcular métricas adicionales
        direction = 1 if ensemble_pred[1] > 0.65 else 0
        confidence = max(ensemble_pred)
        uncertainty = 1 - abs(ensemble_pred[1] - ensemble_pred[0])
        
        return {
            'direction': direction,
            'confidence': confidence,
            'uncertainty': uncertainty,
            'prediction': 'buy' if direction == 1 else 'sell' if confidence > 0.75 else 'hold',
            'ensemble_score': ensemble_pred
        }

    def assess_market_state(self, indicators):
        """Evaluar estado del mercado con más precisión"""
        latest = {k: v.iloc[-1] if isinstance(v, pd.Series) else v for k, v in indicators.items()}
        
        volatility = latest['atr'] / latest['close']
        trend_strength = latest['trend_strength']
        volume_activity = latest['volume_ratio']
        
        # Clasificación más detallada
        if volatility > 0.04:
            if trend_strength > 0.03:
                return 'high_volatility_trending'
            else:
                return 'high_volatility_choppy'
        elif trend_strength > 0.025:
            return 'strong_trending'
        elif trend_strength > 0.015:
            return 'moderate_trending'
        elif volume_activity < 0.7:
            return 'low_volume_consolidation'
        else:
            return 'ranging'

                volatility = data['volatility']
                
                for strategy, signal in signals.items():

                        }

        # Optimizar portafolio y ejecutar operaciones
        await self.optimize_and_execute(all_signals)
        
        # Monitorear posiciones existentes
        await self.monitor_positions()
        
        # Actualizar rendimiento
        self.update_performance()
        
        print(f"Ciclo optimizado completado. Señales generadas: {len(all_signals)}")
    
    def calculate_final_signal_weight(self, signal, market_state, regime, volatility, ml_prediction):
        """Calcular peso final de la señal con múltiples factores"""
        weight = signal['strategy_score']
        
        # Ajuste por estado del mercado
        market_adjustments = {
            'high_volatility_trending': 1.1,
            'strong_trending': 1.2,
            'moderate_trending': 1.1,
            'ranging': 1.0,
            'low_volume_consolidation': 0.9,
            'high_volatility_choppy': 0.8
        }
        
        weight *= market_adjustments.get(market_state, 1.0)
        
        # Ajuste por régimen
        if signal['strategy'] == 'trend' and regime == 'trending':
            weight *= 1.15
        elif signal['strategy'] == 'mean_reversion' and regime == 'ranging':
            weight *= 1.15
        elif signal['strategy'] == 'breakout' and volatility == 'high':
            weight *= 1.1
        
        # Ajuste por predicción de ML
        if ml_prediction and ml_prediction['confidence'] > 0.7:
            if ((signal['type'].endswith('long') and ml_prediction['direction'] == 1) or
                (signal['type'].endswith('short') and ml_prediction['direction'] == 0)):
                weight *= (1 + ml_prediction['confidence'] * 0.2)
        
        return min(weight, 1.0)
    
    async def optimize_and_execute(self, signals):
        """Optimizar portafolio y ejecutar operaciones"""
        if not signals:
            return
        
        # Ordenar señales por peso y estrategia
        signals.sort(key=lambda x: (x['weight'], x['strategy_score']), reverse=True)
        
        # Seleccionar mejores señales con diversificación
        selected_signals = self.select_diversified_signals(signals)
        
        for signal in selected_signals:
            # Calcular tamaño de posición optimizado
            position_size = self.calculate_optimal_position_size(signal)
            
            # Ejecutar operación
            await self.execute_trade(signal, position_size)
    
    def select_diversified_signals(self, signals):
        """Seleccionar señales con diversificación óptima"""
        selected = []
        used_pairs = set()
        used_strategies = {}
        
        for signal in signals:
            pair = signal['pair']
            strategy = signal['strategy']
            
            # Limitar por par
            if pair in used_pairs and len([s for s in selected if s['pair'] == pair]) >= 2:
                continue
            
            # Limitar por estrategia
            if strategy in used_strategies:
                if used_strategies[strategy] >= 2:
                    continue
                used_strategies[strategy] += 1
            else:
                used_strategies[strategy] = 1
            
            # Limitar total de posiciones
            if len(selected) >= self.config.MAX_POSITIONS:
                break
            
            selected.append(signal)
            used_pairs.add(pair)
        
        return selected
    
    def calculate_optimal_position_size(self, signal):
        """Calcular tamaño de posición óptimo"""
        strategy_config = self.config.STRATEGY_RISK[signal['strategy']]
        
        # Ajustar riesgo según confianza
        risk_adjustment = signal['weight']
        base_risk = self.config.RISK_PER_TRADE * risk_adjustment
        
        # Calcular riesgo monetario
        risk_amount = self.capital * base_risk
        
        # Calcular distancia de stop loss
        stop_distance = abs(signal['entry_price'] - signal['stop_loss'])
        
        # Calcular tamaño base
        base_size = risk_amount / stop_distance
        
        # Aplicar apalancamiento según estrategia
        leveraged_size = base_size * strategy_config['leverage']
        
        # Limitar tamaño máximo por posición
        max_size = self.capital * 0.08 / signal['entry_price']
        
        # Ajustar por volatilidad
        volatility_adjustment = 1.0
        if signal['volatility'] == 'high':
            volatility_adjustment = 0.8
        elif signal['volatility'] == 'low':
            volatility_adjustment = 1.1
        
        final_size = min(leveraged_size, max_size) * volatility_adjustment
        
        return max(final_size, 0.001)  # Mínimo 0.001
    
    async def execute_trade(self, signal, position_size):
        """Ejecutar una operación"""
        pair = signal['pair']
        strategy = signal['strategy']
        
        try:
            # Obtener precio actual
            ticker = self.data_fetcher.exchanges['binance'].fetch_ticker(pair)
            current_price = ticker['last']
            
            # Determinar lado
            side = 'buy' if signal['type'].endswith('long') else 'sell'
            
            # Crear ID de posición
            position_id = f"{pair}_{strategy}_{datetime.now().strftime('%Y%m%d%H%M%S')}"
            
            # Registrar posición
            self.positions[position_id] = {
                'id': position_id,
                'pair': pair,
                'strategy': strategy,
                'side': side,
                'entry_price': current_price,
                'size': position_size,
                'stop_loss': signal['stop_loss'],
                'take_profit': signal['take_profit'],
                'weight': signal['weight'],
                'market_state': signal['market_state'],
                'entry_time': datetime.now(),
                'status': 'open',
                'pnl': 0,
                'max_profit': 0
            }
            
            print(f"Operación ejecutada: {side} {position_size:.4f} {pair} @ {current_price} "
                  f"(Confianza: {signal['weight']:.2f})")
            
        except Exception as e:
            print(f"Error ejecutando operación: {e}")
    
    async def monitor_positions(self):
        """Monitorear posiciones abiertas con trailing stop"""
        positions_to_close = []
        
        for position_id, position in self.positions.items():
            if position['status'] != 'open':
                continue
            
            pair = position['pair']
            try:
                # Obtener precio actual
                ticker = self.data_fetcher.exchanges['binance'].fetch_ticker(pair)
                current_price = ticker['last']
                
                # Calcular P&L
                if position['side'] == 'buy':
                    pnl = (current_price - position['entry_price']) * position['size']
                else:
                    pnl = (position['entry_price'] - current_price) * position['size']
                
                position['current_price'] = current_price
                position['pnl'] = pnl
                
                # Actualizar máximo beneficio
                if pnl > position['max_profit']:
                    position['max_profit'] = pnl
                
                # Implementar trailing stop
                if position['max_profit'] > 0:
                    trailing_distance = abs(position['entry_price'] - position['stop_loss']) * 0.5
                    
                    if position['side'] == 'buy':
                        new_stop_loss = current_price - trailing_distance
                        if new_stop_loss > position['stop_loss']:
                            position['stop_loss'] = new_stop_loss
                    else:
                        new_stop_loss = current_price + trailing_distance
                        if new_stop_loss < position['stop_loss']:
                            position['stop_loss'] = new_stop_loss
                
                # Verificar condiciones de cierre
                should_close = False
                close_reason = ""
                
                if position['side'] == 'buy' and current_price <= position['stop_loss']:
                    should_close = True
                    close_reason = "Stop Loss"
                elif position['side'] == 'sell' and current_price >= position['stop_loss']:
                    should_close = True
                    close_reason = "Stop Loss"
                elif position['side'] == 'buy' and current_price >= position['take_profit']:
                    should_close = True
                    close_reason = "Take Profit"
                elif position['side'] == 'sell' and current_price <= position['take_profit']:
                    should_close = True
                    close_reason = "Take Profit"
                
                if should_close:
                    positions_to_close.append((position_id, close_reason))
                
            except Exception as e:
                print(f"Error monitoreando posición {position_id}: {e}")
        
        # Cerrar posiciones
        for position_id, reason in positions_to_close:
            await self.close_position(position_id, reason)
    
    async def close_position(self, position_id, reason):
        """Cerrar una posición y actualizar métricas"""
        if position_id not in self.positions:
            return
        
        position = self.positions[position_id]
        
        # Actualizar estado
        position['status'] = 'closed'
        position['close_time'] = datetime.now()
        position['close_reason'] = reason
        
        # Actualizar capital
        self.capital += position['pnl']
        
        # Actualizar métricas generales
        self.total_trades += 1
        if position['pnl'] > 0:
            self.winning_trades += 1
        self.total_pnl += position['pnl']
        
        # Actualizar métricas por estrategia
        strategy = position['strategy']
        if strategy not in self.strategy_performance:
            self.strategy_performance[strategy] = {
                'trades': 0, 'wins': 0, 'total_pnl': 0
            }
        
        self.strategy_performance[strategy]['trades'] += 1
        if position['pnl'] > 0:
            self.strategy_performance[strategy]['wins'] += 1
        self.strategy_performance[strategy]['total_pnl'] += position['pnl']
        
        # Mover a historial
        self.performance_history.append(position)
        
        # Eliminar de posiciones activas
        del self.positions[position_id]
        
        print(f"Posición cerrada: {position_id} - {reason} - P&L: ${position['pnl']:.2f}")
    
    def update_performance(self):
        """Actualizar métricas de rendimiento detalladas"""
        if self.capital > self.peak_capital:
            self.peak_capital = self.capital
        
        win_rate = self.winning_trades / self.total_trades if self.total_trades > 0 else 0
        current_drawdown = (self.peak_capital - self.capital) / self.peak_capital
        
        print("\n=== Rendimiento Optimizado Quantum Fusion ===")
        print(f"Capital: ${self.capital:,.2f}")
        print(f"Operaciones: {self.total_trades}")
        print(f"Win Rate: {win_rate:.2%}")
        print(f"P&L Total: ${self.total_pnl:,.2f}")
        print(f"Drawdown Actual: {current_drawdown:.2%}")
        print(f"Posiciones abiertas: {len(self.positions)}")
        
        # Rendimiento por estrategia
        print("\nRendimiento por Estrategia:")
        for strategy, perf in self.strategy_performance.items():
            strategy_win_rate = perf['wins'] / perf['trades'] if perf['trades'] > 0 else 0
            print(f"  {strategy}: {perf['trades']} trades, {strategy_win_rate:.2%} win rate, "
                  f"${perf['total_pnl']:,.2f} P&L")
        
        print("============================================\n")
    
    async def run(self, interval_minutes=3):
        """Ejecutar el sistema continuamente con intervalo optimizado"""
        await self.initialize()
        
        print(f"Iniciando Optimized Quantum Fusion Trading System con intervalo de {interval_minutes} minutos")
        
        while True:
            try:
                start_time = time.time()
                
                # Ejecutar ciclo de trading
                await self.execute_trading_cycle()
                
                # Calcular tiempo de espera
                elapsed = time.time() - start_time
                wait_time = max(0, interval_minutes * 60 - elapsed)
                
                print(f"Esperando {wait_time/60:.1f} minutos hasta el siguiente ciclo...")
                await asyncio.sleep(wait_time)
                
            except KeyboardInterrupt:
                print("\nDeteniendo Optimized Quantum Fusion Trading System...")
                break
            except Exception as e:
                print(f"Error en ciclo de trading: {e}")
                await asyncio.sleep(60)
