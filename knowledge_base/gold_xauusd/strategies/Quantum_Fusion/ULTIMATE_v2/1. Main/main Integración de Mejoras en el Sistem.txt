# main.py - Modificaciones para integrar mejoras
class QuantumTradingUltimate:
    def __init__(self, config_path: str = None):
        # ... código existente ...
        
        # Inicializar nuevos componentes
        self.extreme_regime_detector = ExtremeRegimeDetector(self.config)
        self.drift_detector = DriftDetector(self.config)
        self.temporal_validator = TemporalCrossValidator(self.config)
        self.multilayer_decision = MultiLayerDecisionSystem(self.config)
        self.hierarchical_model = HierarchicalEnsembleModel(self.config)
        
        # Estado de gestión de drawdown
        self.drawdown_level = 0
    
    def initialize_components(self):
        # ... código existente ...
        
        # Inicializar nuevos modelos
        self.model = self.hierarchical_model
        self.decision_system = self.multilayer_decision
    
    def generate_signal(self, symbol: str) -> Tuple[str, float, Dict]:
        """Generar señal de trading con todos los filtros y mejoras"""
        try:
            # Preparar características
            features = self.prepare_features(symbol)
            if features.empty or len(features) < 50:
                self.logger.warning(f"No se pudieron preparar características para {symbol}")
                return 'hold', 0.5, {}
            
            # Detectar regímenes extremos
            extreme_regime = self.extreme_regime_detector.detect_extreme_regimes(features)
            regime_recommendations = self.extreme_regime_detector.get_regime_recommendations(extreme_regime)
            
            # Si régimen extremo, aplicar recomendaciones
            if extreme_regime['regime'] in ['extreme', 'high_stress']:
                self.logger.warning(f"Régimen {extreme_regime['regime']} detectado para {symbol}")
                
                # Aplicar acciones recomendadas
                for action in regime_recommendations['actions']:
                    if action['action'] == 'avoid_new_positions' and action['priority'] == 'high':
                        return 'hold', 0.1, {'extreme_regime': extreme_regime}
            
            # Obtener predicción del modelo jerárquico
            latest_features = features.iloc[-1:].copy()
            market_conditions = self.analyze_market_conditions(features)
            
            # Usar sistema de decisión multicapa
            signal, uncertainty = self.multilayer_decision.generate_multilayer_decision(
                latest_features, market_conditions
            )
            
            # Aplicar filtros mejorados
            filtered_signal = self.apply_enhanced_signal_filters(
                signal, uncertainty, market_conditions, extreme_regime
            )
            
            # Calcular fuerza de la señal mejorada
            signal_strength = self.calculate_enhanced_signal_strength(
                filtered_signal, uncertainty, market_conditions, extreme_regime
            )
            
            # Determinar señal final
            if signal_strength > 0.65:
                return 'buy', signal_strength, {
                    'market_conditions': market_conditions,
                    'extreme_regime': extreme_regime,
                    'regime_recommendations': regime_recommendations
                }
            elif signal_strength < 0.35:
                return 'sell', signal_strength, {
                    'market_conditions': market_conditions,
                    'extreme_regime': extreme_regime,
                    'regime_recommendations': regime_recommendations
                }
            else:
                return 'hold', signal_strength, {
                    'market_conditions': market_conditions,
                    'extreme_regime': extreme_regime,
                    'regime_recommendations': regime_recommendations
                }
            
        except Exception as e:
            self.logger.error(f"Error generando señal para {symbol}: {e}")
            return 'hold', 0.5, {}
    
    def apply_enhanced_signal_filters(self, prediction: float, uncertainty: float, 
                                    market_conditions: Dict, extreme_regime: Dict) -> float:
        """Aplicar filtros mejorados a la señal del modelo"""
        try:
            filtered_signal = prediction
            
            # Filtro de confianza
            if uncertainty > 0.3:
                filtered_signal = 0.5
            
            # Filtro de régimen extremo
            regime = extreme_regime.get('regime', 'normal')
            if regime == 'extreme':
                filtered_signal = 0.5  # Mantener posición actual
            elif regime == 'high_stress':
                # Reducir fuerza de señal
                filtered_signal = 0.5 + (filtered_signal - 0.5) * 0.5
            
            # Filtros existentes
            filtered_signal = self.apply_signal_filters(
                prediction, 1 - uncertainty, uncertainty, market_conditions
            )
            
            return filtered_signal
            
        except Exception as e:
            self.logger.error(f"Error aplicando filtros de señal mejorados: {e}")
            return 0.5
    
    def calculate_enhanced_signal_strength(self, signal: float, uncertainty: float, 
                                         market_conditions: Dict, extreme_regime: Dict) -> float:
        """Calcular fuerza final de la señal con mejoras"""
        try:
            # Base: señal filtrada
            strength = signal
            
            # Ajustar por incertidumbre
            if uncertainty > 0.2:
                strength *= (1 - uncertainty)
            
            # Ajustar por régimen extremo
            regime = extreme_regime.get('regime', 'normal')
            if regime == 'extreme':
                strength *= 0.3  # Reducir drásticamente
            elif regime == 'high_stress':
                strength *= 0.6  # Reducir moderadamente
            elif regime == 'moderate_stress':
                strength *= 0.8  # Reducir ligeramente
            
            # Ajustes existentes
            strength = self.calculate_signal_strength(
                signal, 1 - uncertainty, uncertainty, market_conditions
            )
            
            return np.clip(strength, 0, 1)
            
        except Exception as e:
            self.logger.error(f"Error calculando fuerza de señal mejorada: {e}")
            return 0.5
    
    def execute_trade(self, symbol: str, signal: str, signal_strength: float, 
                     market_conditions: Dict):
        """Ejecutar operación con optimización completa y mejoras"""
        if signal == 'hold':
            return
        
        try:
            # Obtener precio actual
            ticker = self.exchange.fetch_ticker(symbol)
            current_price = ticker['last']
            
            # Calcular tamaño de posición
            account_state = self.risk_manager.portfolio_state.copy()
            account_state['positions'] = self.positions
            account_state['current_drawdown'] = \
                self.risk_manager.portfolio_state['current_drawdown']
            account_state['recent_performance'] = \
                self.calculate_recent_performance()
            
            # Preparar características para gestión de riesgo
            features = self.prepare_features(symbol)
            if not features.empty:
                market_features = features.iloc[-1:].copy()
            else:
                market_features = pd.DataFrame()
            
            # Calcular tamaño de posición con gestión de riesgo mejorada
            position_size, risk_factors = \
                self.risk_manager.calculate_position_size(
                    signal_strength, market_features, account_state
                )
            
            # Seleccionar método de ejecución óptimo
            execution_method = self.execution_optimizer.select_optimal_execution_method(
                symbol, position_size, market_conditions
            )
            
            # Ejecutar orden con método seleccionado
            orders, executed_size = \
                self.execution_optimizer.execute_order_with_method(
                    self.exchange, symbol, signal, position_size, 
                    execution_method, market_conditions
                )
            
            if executed_size > 0:
                # Calcular stop-loss adaptativo
                stop_loss = self.risk_manager.calculate_adaptive_stop_loss(
                    {
                        'side': signal,
                        'signal_strength': signal_strength
                    },
                    current_price,
                    market_features
                )
                
                # Calcular take-profit
                take_profit = self.risk_manager.calculate_take_profit(
                    current_price, stop_loss, signal == 'buy'
                )
                
                # Guardar posición
                self.positions[symbol] = {
                    'side': signal,
                    'entry_price': current_price,
                    'quantity': executed_size,
                    'stop_loss': stop_loss,
                    'take_profit': take_profit,
                    'orders': orders,
                    'signal_strength': signal_strength,
                    'entry_time': datetime.now(),
                    'risk_factors': risk_factors,
                    'market_conditions': market_conditions,
                    'execution_method': execution_method
                }
                
                self.logger.info(
                    f"{signal.upper()} ejecutada: {symbol} @ {current_price}, "
                    f"Cantidad: {executed_size}, Método: {execution_method}, "
                    f"SL: {stop_loss}, TP: {take_profit}"
                )
            
        except Exception as e:
            self.logger.error(f"Error ejecutando operación para {symbol}: {e}")
    
    def check_positions(self, symbol: str, current_price: float):
        """Verificar y gestionar posiciones abiertas con mejoras"""
        if symbol not in self.positions:
            return
        
        position = self.positions[symbol]
        
        try:
            # Actualizar trailing stop adaptativo
            features = self.prepare_features(symbol)
            if not features.empty:
                market_features = features.iloc[-1:].copy()
                position = self.risk_manager.update_trailing_stop_adaptive(
                    position, current_price, market_features
                )
                self.positions[symbol] = position
            
            # Verificar stop-loss
            if position['side'] == 'buy':
                if current_price <= position['stop_loss']:
                    self.close_position(symbol, 'stop_loss')
                    return
                
                # Verificar take-profit
                if current_price >= position['take_profit']:
                    self.close_position(symbol, 'take_profit')
                    return
            
            elif position['side'] == 'sell':
                if current_price >= position['stop_loss']:
                    self.close_position(symbol, 'stop_loss')
                    return
                
                # Verificar take-profit
                if current_price <= position['take_profit']:
                    self.close_position(symbol, 'take_profit')
                    return
            
        except Exception as e:
            self.logger.error(f"Error verificando posición para {symbol}: {e}")
    
    def update_portfolio_state(self):
        """Actualizar estado completo del portafolio con gestión de drawdown"""
        try:
            # Obtener precios actuales
            current_prices = {}
            for symbol in self.positions.keys():
                try:
                    ticker = self.exchange.fetch_ticker(symbol)
                    current_prices[symbol] = ticker['last']
                except:
                    current_prices[symbol] = \
                        self.positions[symbol]['entry_price']
            
            # Actualizar estado del portafolio
            portfolio_state = \
                self.risk_manager.update_portfolio_state(self.positions, current_prices)
            
            # Verificar niveles de drawdown
            self.drawdown_level = self.risk_manager.check_drawdown_levels(portfolio_state)
            
            # Verificar límites de riesgo
            risk_alerts = \
                self.risk_manager.check_risk_limits(portfolio_state)
            
            for alert in risk_alerts:
                self.logger.warning(f"Alerta de riesgo: {alert['message']} "
                                   f"(Severidad: {alert['severity']})")
            
            return portfolio_state
            
        except Exception as e:
            self.logger.error(f"Error actualizando estado del portafolio: {e}")
            return {}
    
    def train_model(self, symbol: str) -> float:
        """Entrenar o actualizar el modelo con validación temporal"""
        try:
            self.logger.info(f"Entrenando modelo para {symbol}...")
            
            # Preparar características
            features = self.prepare_features(symbol)
            if features.empty or len(features) < 100:
                self.logger.error("No se pudieron preparar suficientes características")
                return 0.0
            
            # Preparar etiquetas
            labels = self.prepare_labels(features)
            
            # Validación cruzada temporal
            if self.config.get('validation', {}).get('enabled', False):
                validation_results = self.temporal_validator.walk_forward_validation(
                    pd.concat([features, labels.rename('target')], axis=1),
                    HierarchicalEnsembleModel,
                    {'config': self.config}
                )
                
                if validation_results:
                    accuracy = validation_results['aggregated_metrics']['accuracy']
                    self.logger.info(f"Validación walk-forward para {symbol}: "
                                    f"accuracy={accuracy:.4f}")
            
            # Entrenar modelo jerárquico
            self.hierarchical_model.train_level1_models(features, labels)
            self.hierarchical_model.train_meta_model(features, labels)
            
            # Obtener condiciones de mercado para entrenamiento de decisión
            market_conditions_list = []
            for i in range(len(features)):
                if i < 20:  # Necesitamos suficientes datos previos
                    market_conditions_list.append({})
                else:
                    market_conditions_list.append(
                        self.analyze_market_conditions(features.iloc[:i+1])
                    )
            
            market_conditions_df = pd.DataFrame(market_conditions_list)
            self.hierarchical_model.train_decision_model(
                features, labels, market_conditions_df
            )
            
            # Guardar modelo
            self.hierarchical_model.save_model(f'{symbol}_hierarchical')
            
            # Registrar actualización
            self.last_model_update = datetime.now()
            self.model_update_history.append({
                'timestamp': self.last_model_update,
                'symbol': symbol,
                'accuracy': accuracy if 'accuracy' in locals() else 0.0,
                'features_count': len(features.columns)
            })
            
            self.logger.info(f"Modelo entrenado para {symbol}")
            return accuracy if 'accuracy' in locals() else 0.0
            
        except Exception as e:
            self.logger.error(f"Error entrenando modelo para {symbol}: {e}")
            return 0.0
    
    def run(self):
        """Ejecutar bot de trading principal con mejoras"""
        try:
            self.logger.info("Iniciando Quantum Trading Ultimate con mejoras...")
            self.is_running = True
            
            # Entrenar modelos iniciales
            for symbol in self.config['symbols']:
                try:
                    self.train_model(symbol)
                except Exception as e:
                    self.logger.error(f"Error entrenando modelo para {symbol}: {e}")
            
            # Bucle principal de trading
            while self.is_running:
                try:
                    # Verificar si se debe reentrenar por drift
                    if self.drift_detector.should_retrain():
                        self.logger.info("Detectada necesidad de reentrenamiento por drift")
                        for symbol in self.config['symbols']:
                            try:
                                self.train_model(symbol)
                            except Exception as e:
                                self.logger.error(f"Error reentrenando modelo para {symbol}: {e}")
                    
                    # Actualizar modelos periódicamente
                    if self.should_update_models():
                        for symbol in self.config['symbols']:
                            try:
                                self.train_model(symbol)
                            except Exception as e:
                                self.logger.error(f"Error actualizando modelo para {symbol}: {e}")
                    
                    # Procesar cada símbolo
                    for symbol in self.config['symbols']:
                        try:
                            # Obtener precio actual
                            ticker = self.exchange.fetch_ticker(symbol)
                            current_price = ticker['last']
                            
                            # Generar señal
                            signal, signal_strength, context = self.generate_signal(symbol)
                            
                            # Ejecutar operación si hay señal
                            if signal != 'hold':
                                self.execute_trade(symbol, signal, signal_strength, context)
                            
                            # Verificar posiciones abiertas
                            self.check_positions(symbol, current_price)
                            
                            # Actualizar estado del portafolio
                            portfolio_state = self.update_portfolio_state()
                            
                            # Detectar drift en predicciones
                            if hasattr(self, 'last_predictions') and hasattr(self, 'last_actuals'):
                                drift_detected = self.drift_detector.detect_drift(
                                    self.last_predictions, self.last_actuals
                                )
                                if drift_detected:
                                    self.logger.warning(f"Drift detectado para {symbol}")
                            
                            self.logger.info(
                                f"{symbol}: {signal} (Fuerza: {signal_strength:.2f}), "
                                f"Drawdown: {self.drawdown_level}, "
                                f"Portfolio: ${portfolio_state.get('total_value', 0):.2f}, "
                                f"PnL Diario: {portfolio_state.get('daily_pnl', 0):.2%}"
                            )
                            
                        except Exception as e:
                            self.logger.error(f"Error procesando {symbol}: {e}")
                    
                    # Esperar antes del siguiente ciclo
                    time.sleep(self.config['trading']['interval'])
                
                except KeyboardInterrupt:
                    self.logger.info("Bot detenido por el usuario")
                    self.is_running = False
                    break
                except Exception as e:
                    self.logger.error(f"Error en bucle principal: {e}")
                    time.sleep(60)
            
        except Exception as e:
            self.logger.error(f"Error fatal en ejecución: {e}")
