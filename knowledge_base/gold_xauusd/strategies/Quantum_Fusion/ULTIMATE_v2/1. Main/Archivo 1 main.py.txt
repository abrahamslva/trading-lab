    def get_initial_capital(self) -> float:

    def get_market_data(self, symbol: str, timeframe: str = '1h', limit: int = 500) -> pd.DataFrame:
        """Obtener datos de mercado del exchange"""
        try:
            ohlcv = self.exchange.fetch_ohlcv(symbol, timeframe, limit=limit)
            df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
            df['timestamp'] = pd.to_datetime(df['timestamp'], unit='ms')
            df.set_index('timestamp', inplace=True)
            return df
        except Exception as e:
            self.logger.error(f"Error obteniendo datos de mercado para {symbol}: {e}")
            return pd.DataFrame()
    
    def prepare_features(self, symbol: str, timeframe: str = '1h') -> pd.DataFrame:
        """Preparar características para el modelo"""
        try:
            # Obtener datos de mercado
            df = self.get_market_data(symbol, timeframe)
            if df.empty:
                self.logger.warning(f"No se pudieron obtener datos para {symbol}")
                return pd.DataFrame()
            
            # Construir características técnicas
            features = self.indicators.build_advanced_features(df)
            
            # Obtener datos de sentimiento
            sentiment_data = self.update_sentiment_data(symbol)
            
            # Unir con sentimiento
            features = features.join(sentiment_data, how='left')
            features['sentiment'] = features['sentiment'].fillna(0)
            features['sentiment_volatility'] = features['sentiment_volatility'].fillna(0)
            features['sentiment_momentum'] = features['sentiment_momentum'].fillna(0)
            
            # Obtener señales de APIs premium
            premium_signals = self.get_premium_signals(symbol)
            
            # Integrar señales premium
            features = self.integrate_premium_signals(features, premium_signals)
            
            # Añadir características de régimen de mercado
            regime = self.market_regime_detector.detect_market_regime_advanced(df)
            features['market_regime'] = regime.reindex(features.index, fill_value=0)
            
            self.logger.debug(f"Características preparadas para {symbol}: {len(features)} filas")
            return features
            
        except Exception as e:
            self.logger.error(f"Error preparando características para {symbol}: {e}")
            return pd.DataFrame()
    
    def update_sentiment_data(self, symbol: str) -> pd.DataFrame:

        try:
            base_currency = symbol.split('/')[0]
            
            # Obtener sentimiento de diferentes fuentes
            twitter_sentiment = self.sentiment.get_twitter_sentiment(base_currency, count=100)
            reddit_sentiment = self.sentiment.get_reddit_sentiment(f'CryptoCurrency', limit=50)
            news_sentiment = self.sentiment.get_news_sentiment(base_currency, days=3)
            onchain_sentiment = self.sentiment.get_onchain_sentiment(symbol)
            
            # Agregar sentimiento
            sentiment_df = self.sentiment.aggregate_sentiment([

            return sentiment_df
            
        except Exception as e:
            self.logger.error(f"Error actualizando sentimiento para {symbol}: {e}")
            return pd.DataFrame({'sentiment': [0], 'sentiment_volatility': [0], 'sentiment_momentum': [0]})
    
    def get_premium_signals(self, symbol: str) -> Dict:
        """Obtener señales de todas las APIs premium"""
        premium_signals = {}
        
        # Mapear símbolos a IDs de APIs
        symbol_mapping = {
            'BTC/USDT': {
                'glassnode': 'BTC',
                'cryptoquant': 'btc',
                'lunarcrush': 'BTC',
                'kaiko': {'exchange': 'binance', 'pair': 'BTCUSDT'}
            },
            'ETH/USDT': {
                'glassnode': 'ETH',
                'cryptoquant': 'eth',
                'lunarcrush': 'ETH',
                'kaiko': {'exchange': 'binance', 'pair': 'ETHUSDT'}
            },
            'BNB/USDT': {
                'glassnode': 'BNB',
                'cryptoquant': 'bnb',
                'lunarcrush': 'BNB',
                'kaiko': {'exchange': 'binance', 'pair': 'BNBUSDT'}
            }
        }
        
        if symbol not in symbol_mapping:
            return premium_signals
        
        mapping = symbol_mapping[symbol]
        
        # Obtener señales de cada API premium
        for api_name, analyzer in self.premium_apis.items():
            try:
                if api_name == 'glassnode' and 'glassnode' in mapping:
                    whale_data = analyzer.get_whale_intelligence(mapping['glassnode'])
                    premium_signals['glassnode'] = analyzer.calculate_whale_signal(whale_data)
                
                elif api_name == 'cryptoquant' and 'cryptoquant' in mapping:
                    flow_data = analyzer.get_exchange_flow_data(mapping['cryptoquant'])
                    premium_signals['cryptoquant'] = analyzer.generate_flow_signals(flow_data)
                
                elif api_name == 'lunarcrush' and 'lunarcrush' in mapping:
                    social_data = analyzer.get_social_sentiment(mapping['lunarcrush'])
                    premium_signals['lunarcrush'] = analyzer.calculate_social_signals(social_data)
                
                elif api_name == 'kaiko' and 'kaiko' in mapping:
                    kaiko_mapping = mapping['kaiko']
                    order_book = analyzer.get_order_book_data(
                        kaiko_mapping['exchange'], kaiko_mapping['pair']
                    )
                    metrics = analyzer.calculate_liquidity_metrics(order_book)
                    premium_signals['kaiko'] = analyzer.generate_liquidity_signals(metrics)
                
            except Exception as e:
                self.logger.warning(f"Error obteniendo señales de {api_name}: {e}")
                premium_signals[api_name] = {}
        
        return premium_signals
    
    def integrate_premium_signals(self, features: pd.DataFrame, premium_signals: Dict) -> pd.DataFrame:
        """Integrar señales de APIs premium en las características"""
        try:
            # Definir pesos para cada tipo de señal
            weights = {
                'glassnode': 0.25,  # On-chain whale activity
                'cryptoquant': 0.25,  # Exchange flows
                'lunarcrush': 0.25,  # Social sentiment
                'kaiko': 0.25  # Liquidity metrics
            }
            
            # Calcular puntuación combinada
            combined_score = 0
            total_weight = 0
            
            for api_name, signals in premium_signals.items():
                if api_name in weights and signals:
                    # Calcular puntuación promedio de las señales de esta API
                    api_score = np.mean(list(signals.values())) if signals else 0
                    weight = weights[api_name]
                    
                    combined_score += api_score * weight
                    total_weight += weight
                    
                    # Añadir señales individuales como características
                    for signal_name, signal_value in signals.items():
                        features[f'{api_name}_{signal_name}'] = signal_value
            
            # Normalizar puntuación
            if total_weight > 0:
                combined_score /= total_weight
            
            features['premium_combined_score'] = combined_score
            features['premium_confidence'] = min(abs(combined_score) * 2, 1.0)
            
            self.logger.debug(f"Señales premium integradas: score={combined_score:.3f}, confidence={features['premium_confidence'].iloc[-1]:.3f}")
            
            return features
            
        except Exception as e:
            self.logger.error(f"Error integrando señales premium: {e}")
            features['premium_combined_score'] = 0
            features['premium_confidence'] = 0.5
            return features
    
    def prepare_labels(self, df: pd.DataFrame) -> pd.Series:
        """Preparar etiquetas multi-horizonte para el modelo"""
        try:
            labels = {}
            
            # Múltiples horizontes temporales
            for h in [3, 5, 8, 12]:
                future_returns = df['close'].shift(-h).pct_change()
                labels[f'label_{h}h'] = (future_returns > 0).astype(int)
            
            # Combinar etiquetas con pesos
            combined_label = (
                0.4 * labels['label_5h'] +
                0.3 * labels['label_3h'] +
                0.2 * labels['label_8h'] +
                0.1 * labels['label_12h']
            )
            
            return (combined_label > 0.5).astype(int)
            
        except Exception as e:
            self.logger.error(f"Error preparando etiquetas: {e}")
            return pd.Series(0, index=df.index)
    
    def generate_signal(self, symbol: str) -> Tuple[str, float, Dict]:
        """Generar señal de trading con todos los filtros y mejoras"""
        try:
            # Preparar características
            features = self.prepare_features(symbol)
            if features.empty or len(features) < 50:
                self.logger.warning(f"No se pudieron preparar características para {symbol}")
                return 'hold', 0.5, {}
            
            # Detectar ciclo de mercado
            market_cycle = self.market_cycle_detector.detect_market_cycle(features)
            cycle_strategy = self.market_cycle_detector.get_cycle_strategy(market_cycle)
            
            # Detectar regímenes extremos
            extreme_regime = self.extreme_regime_detector.detect_extreme_regimes(features)
            regime_recommendations = self.extreme_regime_detector.get_regime_recommendations(extreme_regime)
            
            # Si régimen extremo, aplicar recomendaciones
            if extreme_regime['regime'] in ['extreme', 'high_stress']:
                self.logger.warning(f"Régimen {extreme_regime['regime']} detectado para {symbol}")
                
                for action in regime_recommendations['actions']:
                    if action['action'] == 'avoid_new_positions' and action['priority'] == 'high':
                        return 'hold', 0.1, {
                            'extreme_regime': extreme_regime,
                            'market_cycle': market_cycle
                        }
            
            # Obtener predicciones del ensemble multi-temporal
            latest_features = features.iloc[-1:].copy()
            market_conditions = self.analyze_market_conditions(features)
            
            # Predicción del ensemble multi-temporal
            temporal_prediction, temporal_confidence = self.multi_temporal_ensemble.predict_with_temporal_ensemble(latest_features)
            
            # Predicción del modelo jerárquico
            hierarchical_prediction, hierarchical_uncertainty, hierarchical_confidence = self.hierarchical_model.predict_final_decision(latest_features, market_conditions)
            
            # Combinar predicciones
            model_predictions = {
                'multi_temporal': (temporal_prediction, temporal_confidence),
                'hierarchical': (hierarchical_prediction, hierarchical_confidence)
            }
            
            # Generar decisión probabilística
            probabilistic_decision = self.probabilistic_decision.generate_probabilistic_decision(
                latest_features, model_predictions
            )
            
            # Aplicar filtros mejorados
            filtered_signal = self.apply_enhanced_signal_filters(
                probabilistic_decision['decision'],
                probabilistic_decision['uncertainty'],
                market_conditions,
                extreme_regime,
                market_cycle
            )
            
            # Calcular fuerza de la señal mejorada
            signal_strength = self.calculate_enhanced_signal_strength(
                filtered_signal,
                probabilistic_decision['uncertainty'],
                market_conditions,
                extreme_regime,
                market_cycle,
                cycle_strategy
            )
            
            # Determinar señal final
            if signal_strength > 0.65:
                return 'buy', signal_strength, {
                    'market_conditions': market_conditions,
                    'extreme_regime': extreme_regime,
                    'market_cycle': market_cycle,
                    'cycle_strategy': cycle_strategy,
                    'probabilistic_decision': probabilistic_decision,
                    'regime_recommendations': regime_recommendations
                }
            elif signal_strength < 0.35:
                return 'sell', signal_strength, {
                    'market_conditions': market_conditions,
                    'extreme_regime': extreme_regime,
                    'market_cycle': market_cycle,
                    'cycle_strategy': cycle_strategy,
                    'probabilistic_decision': probabilistic_decision,
                    'regime_recommendations': regime_recommendations
                }
            else:
                return 'hold', signal_strength, {
                    'market_conditions': market_conditions,
                    'extreme_regime': extreme_regime,
                    'market_cycle': market_cycle,
                    'cycle_strategy': cycle_strategy,
                    'probabilistic_decision': probabilistic_decision,
                    'regime_recommendations': regime_recommendations
                }
            
        except Exception as e:
            self.logger.error(f"Error generando señal para {symbol}: {e}")
            return 'hold', 0.5, {}
    
    def analyze_market_conditions(self, features: pd.DataFrame) -> Dict:

        except Exception as e:

    def apply_signal_filters(self, prediction: float, confidence: float, uncertainty: float, market_conditions: Dict) -> float:
        """Aplicar filtros a la señal del modelo"""
        try:
            filtered_signal = prediction
            
            # Filtro de confianza
            if confidence < 0.6:
                filtered_signal = 0.5
            
            # Filtro de incertidumbre
            if uncertainty > 0.3:
                filtered_signal = 0.5
            
            # Filtro de régimen
            regime = market_conditions.get('market_regime', 0)
            if regime == -1:  # Régimen bajista
                filtered_signal = min(filtered_signal, 0.4)
            elif regime == 1:  # Régimen alcista
                filtered_signal = max(filtered_signal, 0.6)
            
            # Filtro de extremos técnicos
            rsi = market_conditions.get('rsi', 50)
            if rsi > 80:
                filtered_signal = min(filtered_signal, 0.3)
            elif rsi < 20:
                filtered_signal = max(filtered_signal, 0.7)
            
            # Filtro de sentimiento extremo
            sentiment = market_conditions.get('sentiment', 0)
            if sentiment > 0.8:
                filtered_signal = max(filtered_signal, 0.6)
            elif sentiment < -0.8:
                filtered_signal = min(filtered_signal, 0.4)
            
            # Filtro de señales premium
            premium_score = market_conditions.get('premium_score', 0)
            premium_confidence = market_conditions.get('premium_confidence', 0.5)
            
            if premium_confidence > 0.8:  # Alta confianza en señales premium
                if premium_score > 0.3:
                    filtered_signal = max(filtered_signal, 0.7)
                elif premium_score < -0.3:
                    filtered_signal = min(filtered_signal, 0.3)
            
            return filtered_signal
            
        except Exception as e:
            self.logger.error(f"Error aplicando filtros de señal: {e}")
            return 0.5
    
    def apply_enhanced_signal_filters(self, prediction: float, uncertainty: float,
                                    market_conditions: Dict, extreme_regime: Dict,
                                    market_cycle: Dict) -> float:
        """Aplicar filtros mejorados a la señal del modelo"""
        try:
            filtered_signal = prediction
            
            # Filtro de confianza
            if uncertainty > 0.3:
                filtered_signal = 0.5
            
            # Filtro de régimen extremo
            regime = extreme_regime.get('regime', 'normal')
            if regime == 'extreme':
                filtered_signal = 0.5  # Mantener posición actual
            elif regime == 'high_stress':
                # Reducir fuerza de señal
                filtered_signal = 0.5 + (filtered_signal - 0.5) * 0.5
            
            # Filtro de ciclo de mercado
            cycle_phase = market_cycle.get('phase', 'unknown')
            cycle_confidence = market_cycle.get('confidence', 0.5)
            
            if cycle_confidence > 0.7:  # Solo aplicar si alta confianza en ciclo
                if cycle_phase == 'accumulation' and filtered_signal < 0.5:
                    # En acumulación, favorecer señales de compra
                    filtered_signal = max(filtered_signal, 0.4)
                elif cycle_phase == 'distribution' and filtered_signal > 0.5:
                    # En distribución, favorecer señales de venta
                    filtered_signal = min(filtered_signal, 0.6)
                elif cycle_phase == 'markdown':
                    # En markdown, reducir fuerza de compras
                    if filtered_signal > 0.5:
                        filtered_signal = 0.5 + (filtered_signal - 0.5) * 0.5
            
            # Filtros existentes
            filtered_signal = self.apply_signal_filters(
                prediction, 1 - uncertainty, uncertainty, market_conditions
            )
            
            return filtered_signal
            
        except Exception as e:
            self.logger.error(f"Error aplicando filtros de señal mejorados: {e}")
            return 0.5
    
    def calculate_signal_strength(self, signal: float, confidence: float, uncertainty: float, market_conditions: Dict) -> float:
        """Calcular fuerza final de la señal"""
        try:
            # Base: señal filtrada
            strength = signal
            
            # Ajustar por confianza
            if confidence > 0.8:
                strength *= 1.2
            elif confidence < 0.6:
                strength *= 0.8
            
            # Ajustar por incertidumbre
            if uncertainty > 0.2:
                strength *= (1 - uncertainty)
            
            # Ajustar por condiciones de mercado
            volatility = market_conditions.get('volatility', 0)
            if volatility > 0.05:  # Alta volatilidad
                strength *= 0.8
            
            # Ajustar por volumen
            volume_ratio = market_conditions.get('volume_ratio', 1)
            if volume_ratio > 1.5:  # Alto volumen confirma la señal
                strength *= 1.1
            
            # Ajustar por señales premium
            premium_confidence = market_conditions.get('premium_confidence', 0.5)
            if premium_confidence > 0.7:
                strength *= 1.15
            
            return np.clip(strength, 0, 1)
            
        except Exception as e:
            self.logger.error(f"Error calculando fuerza de señal: {e}")
            return 0.5
    
    def calculate_enhanced_signal_strength(self, signal: float, uncertainty: float,
                                         market_conditions: Dict, extreme_regime: Dict,
                                         market_cycle: Dict, cycle_strategy: Dict) -> float:
        """Calcular fuerza final de la señal con mejoras"""
        try:
            # Base: señal filtrada
            strength = signal
            
            # Ajustar por incertidumbre
            if uncertainty > 0.2:
                strength *= (1 - uncertainty)
            
            # Ajustar por régimen extremo
            regime = extreme_regime.get('regime', 'normal')
            if regime == 'extreme':
                strength *= 0.3  # Reducir drásticamente
            elif regime == 'high_stress':
                strength *= 0.6  # Reducir moderadamente
            elif regime == 'moderate_stress':
                strength *= 0.8  # Reducir ligeramente
            
            # Ajustar por ciclo de mercado
            cycle_phase = market_cycle.get('phase', 'unknown')
            cycle_confidence = market_cycle.get('confidence', 0.5)
            
            if cycle_confidence > 0.7:  # Solo aplicar si alta confianza en ciclo
                position_size_factor = cycle_strategy.get('position_size', 0.5)
                strength *= position_size_factor
            
            # Ajustes existentes
            strength = self.calculate_signal_strength(
                signal, 1 - uncertainty, uncertainty, market_conditions
            )
            
            return np.clip(strength, 0, 1)
            
        except Exception as e:
            self.logger.error(f"Error calculando fuerza de señal mejorada: {e}")
            return 0.5
    
    def execute_trade(self, symbol: str, signal: str, signal_strength: float, market_conditions: Dict):
        """Ejecutar operación con todas las optimizaciones"""
        if signal == 'hold':
            return
        
        try:
            # Obtener precio actual y order book
            ticker = self.exchange.fetch_ticker(symbol)
            current_price = ticker['last']
            order_book = self.exchange.fetch_order_book(symbol, limit=20)
            
            # Análisis de microestructura para timing
            microstructure_analysis = self.microstructure_analyzer.analyze_order_flow(symbol, order_book)
            
            # Si no es timing óptimo, esperar
            if not microstructure_analysis['optimal_timing'] and microstructure_analysis['confidence'] > 0.7:
                self.logger.info(f"Timing no óptimo para {symbol}, posponiendo ejecución")
                return
            
            # Calcular tamaño base
            account_state = self.risk_manager.portfolio_state.copy()
            account_state['positions'] = self.positions
            account_state['current_drawdown'] = self.risk_manager.portfolio_state['current_drawdown']
            account_state['recent_performance'] = self.calculate_recent_performance()
            
            features = self.prepare_features(symbol)
            base_position_size, risk_factors = self.risk_manager.calculate_position_size(
                signal_strength, features.iloc[-1:] if not features.empty else pd.DataFrame(), account_state
            )
            
            # Calcular tamaño dinámico óptimo
            optimal_size = self.dynamic_sizing.calculate_optimal_order_size(
                symbol, base_position_size, order_book, market_conditions
            )
            
            # Ajustar tamaño por decisión probabilística
            if 'probabilistic_decision' in market_conditions:
                final_size = self.probabilistic_decision.calculate_position_size_probabilistic(
                    market_conditions['probabilistic_decision'], optimal_size
                )
            else:
                final_size = optimal_size
            
            # Seleccionar método de ejecución
            execution_method = self.execution_optimizer.select_optimal_execution_method(
                symbol, final_size, market_conditions
            )
            
            # Ejecutar orden
            orders, executed_size = self.execution_optimizer.execute_order_with_method(
                self.exchange, symbol, signal, final_size, execution_method, market_conditions
            )
            
            if executed_size > 0:
                # Calcular stops adaptativos
                stop_loss = self.risk_manager.calculate_adaptive_stop_loss(
                    {'side': signal, 'signal_strength': signal_strength},
                    current_price, features.iloc[-1:] if not features.empty else pd.DataFrame()
                )
                take_profit = self.risk_manager.calculate_take_profit(
                    current_price, stop_loss, signal == 'buy'
                )
                
                # Guardar posición
                self.positions[symbol] = {
                    'side': signal,
                    'entry_price': current_price,
                    'quantity': executed_size,
                    'stop_loss': stop_loss,
                    'take_profit': take_profit,
                    'orders': orders,
                    'signal_strength': signal_strength,
                    'entry_time': datetime.now(),
                    'risk_factors': risk_factors,
                    'market_conditions': market_conditions,
                    'execution_method': execution_method,
                    'microstructure_analysis': microstructure_analysis
                }
                
                self.logger.info(
                    f"{signal.upper()} ejecutada: {symbol} @ {current_price}, "
                    f"Cantidad: {executed_size}, Método: {execution_method}"
                )
                
                # Guardar predicción para detección de drift
                if 'probabilistic_decision' in market_conditions:
                    self.last_predictions.append(market_conditions['probabilistic_decision']['decision'])
                
                # Actualizar importancia de características
                if not features.empty:
                    self.feature_monitor.update_feature_importance(
                        self.model, features, list(features.columns)
                    )
                
                # Verificar si se necesita cobertura
                self.check_hedge_requirements()
            
        except Exception as e:
            self.logger.error(f"Error ejecutando operación para {symbol}: {e}")
    
    def check_hedge_requirements(self):
        """Verificar si se requiere cobertura"""
        try:
            # Obtener precios actuales
            current_prices = {}
            for symbol in self.positions.keys():
                try:
                    ticker = self.exchange.fetch_ticker(symbol)
                    current_prices[symbol] = ticker['last']
                except:
                    current_prices[symbol] = self.positions[symbol]['entry_price']
            
            # Verificar cobertura delta
            if self.config['hedging']['delta_hedging']['enabled']:
                self.hedging_manager.execute_delta_hedge(self.positions, current_prices)
            
            # Verificar cobertura de cola
            if self.config['hedging']['tail_hedging']['enabled']:
                market_conditions = {}
                for symbol in self.positions.keys():
                    features = self.prepare_features(symbol)
                    if not features.empty:
                        market_conditions[symbol] = self.analyze_market_conditions(features)
                
                self.hedging_manager.execute_tail_hedge(market_conditions)
            
        except Exception as e:
            self.logger.error(f"Error verificando requerimientos de cobertura: {e}")
    
    def calculate_recent_performance(self, days: int = 7) -> float:
        """Calcular rendimiento reciente"""
        if not self.performance_history:
            return 0.0
        
        recent_performance = [
            p['daily_return'] for p in self.performance_history[-days:]
            if 'daily_return' in p
        ]
        
        if not recent_performance:
            return 0.0
        
        return np.mean(recent_performance)
    
    def check_positions(self, symbol: str, current_price: float):
        """Verificar y gestionar posiciones abiertas"""
        if symbol not in self.positions:
            return
        
        position = self.positions[symbol]
        
        try:
            # Actualizar trailing stop
            features = self.prepare_features(symbol)
            if not features.empty:
                market_features = features.iloc[-1:].copy()
                position = self.risk_manager.update_trailing_stop_adaptive(
                    position, current_price, market_features
                )
                self.positions[symbol] = position
            
            # Verificar stop-loss
            if position['side'] == 'buy':
                if current_price <= position['stop_loss']:
                    self.close_position(symbol, 'stop_loss')
                    return
                
                # Verificar take-profit
                if current_price >= position['take_profit']:
                    self.close_position(symbol, 'take_profit')
                    return
            
            elif position['side'] == 'sell':
                if current_price >= position['stop_loss']:
                    self.close_position(symbol, 'stop_loss')
                    return
                
                # Verificar take-profit
                if current_price <= position['take_profit']:
                    self.close_position(symbol, 'take_profit')
                    return
            
        except Exception as e:
            self.logger.error(f"Error verificando posición para {symbol}: {e}")
    
    def close_position(self, symbol: str, reason: str):
        """Cerrar posición con tracking completo"""
        if symbol not in self.positions:
            return
        
        position = self.positions[symbol]
        
        try:
            if position['side'] == 'buy':
                order = self.exchange.create_market_sell_order(symbol, position['quantity'])
            else:
                order = self.exchange.create_market_buy_order(symbol, position['quantity'])
            
            # Calcular PnL
            current_price = order['price']
            if position['side'] == 'buy':
                pnl = (current_price - position['entry_price']) / position['entry_price']
            else:
                pnl = (position['entry_price'] - current_price) / position['entry_price']
            
            # Guardar actual para detección de drift
            self.last_actuals.append(1 if pnl > 0 else 0)
            
            # Limitar historial
            max_history = 100
            if len(self.last_predictions) > max_history:
                self.last_predictions = self.last_predictions[-max_history:]
            if len(self.last_actuals) > max_history:
                self.last_actuals = self.last_actuals[-max_history:]
            
            # Registrar cierre
            self.logger.info(f"Posición cerrada - {symbol}: {reason}, PnL: {pnl:.2%}")
            
            # Actualizar rendimiento
            self.indicators.update_indicator_performance(symbol, position['side'], pnl)
            self.sentiment.update_sentiment_performance(symbol, pnl)
            
            # Actualizar rendimiento de modelos temporales
            if 'probabilistic_decision' in position['market_conditions']:
                pred = position['market_conditions']['probabilistic_decision']['decision']
                actual = 1 if pnl > 0 else 0
                
                # Actualizar rendimiento de cada ventana temporal
                for window_name in self.multi_temporal_ensemble.temporal_windows.keys():
                    for model_name in self.multi_temporal_ensemble.models[window_name].keys():
                        self.multi_temporal_ensemble.update_model_performance(
                            window_name, model_name, pred, actual
                        )
            
            # Eliminar posición
            del self.positions[symbol]
            
            # Actualizar historial de rendimiento
            self.performance_history.append({
                'timestamp': datetime.now(),
                'symbol': symbol,
                'reason': reason,
                'pnl': pnl,
                'holding_period': (datetime.now() - position['entry_time']).total_seconds() / 3600,
                'signal_type': position['side'],
                'market_conditions': position['market_conditions']
            })
            
        except Exception as e:
            self.logger.error(f"Error cerrando posición para {symbol}: {e}")
    
    def update_portfolio_state(self):

                    ticker = self.exchange.fetch_ticker(symbol)
                    current_prices[symbol] = ticker['last']
                except:
                    current_prices[symbol] = self.positions[symbol]['entry_price']
            
            # Actualizar estado del portafolio
            portfolio_state = self.risk_manager.update_portfolio_state(self.positions, current_prices)
            
            # Verificar niveles de drawdown
            self.drawdown_level = self.drawdown_manager.check_drawdown_levels(portfolio_state)

                    df = self.get_market_data(symbol, '1h', 100)
                    if not df.empty:
                        market_data[symbol] = df

                    pass
            
            if market_data:

                if correlation_matrix is not None:

                    portfolio_state['avg_correlation'] = avg_correlation

    def get_historical_data(self, symbol: str, start_date: str, end_date: str) -> pd.DataFrame:
        """Obtener datos históricos para backtesting"""
        try:
            # Calcular número de velas necesarias
            start_dt = datetime.strptime(start_date, '%Y-%m-%d')
            end_dt = datetime.strptime(end_date, '%Y-%m-%d')
            days_diff = (end_dt - start_dt).days
            
            # Obtener datos en chunks de 1000 velas
            all_data = []
            current_time = end_dt
            
            while current_time > start_dt:
                limit = min(1000, int(days_diff * 24))  # Máximo 1000 velas
                
                ohlcv = self.exchange.fetch_ohlcv(
                    symbol, '1h', limit=limit,
                    since=int(current_time.timestamp() * 1000)
                )
                
                if ohlcv:
                    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
                    df['timestamp'] = pd.to_datetime(df['timestamp'], unit='ms')
                    all_data.append(df)
                
                current_time = current_time - timedelta(days=30)  # Retroceder 30 días
            
            if all_data:
                result = pd.concat(all_data, ignore_index=True)
                result = result[result['timestamp'] >= start_dt]
                result = result[result['timestamp'] <= end_dt]
                result.set_index('timestamp', inplace=True)
                return result.sort_index()
            
            return pd.DataFrame()
            
        except Exception as e:
            self.logger.error(f"Error obteniendo datos históricos: {e}")
            return pd.DataFrame()
    
    def calculate_strategy_returns(self, data: pd.DataFrame, symbol: str) -> pd.Series:
        """Calcular retornos de la estrategia para simulación Monte Carlo"""
        try:
            if data.empty:
                return pd.Series()
            
            # Simular generación de señales (simplificado para backtesting)
            signals = pd.Series(0, index=data.index)
            
            # Lógica simplificada de señales para backtesting
            for i in range(20, len(data)):
                latest_data = data.iloc[i-20:i]
                
                # Calcular algunos indicadores básicos
                returns = latest_data['close'].pct_change()
                rsi = self.calculate_rsi(latest_data['close'])
                
                # Generar señal basada en condiciones
                if rsi.iloc[-1] < 30 and returns.iloc[-1] < -0.02:
                    signals.iloc[i] = 1  # Señal de compra
                elif rsi.iloc[-1] > 70 and returns.iloc[-1] > 0.02:
                    signals.iloc[i] = -1  # Señal de venta
            
            # Calcular retornos de la estrategia
            strategy_returns = data['close'].pct_change() * signals.shift(1)
            
            return strategy_returns.fillna(0)
            
        except Exception as e:
            self.logger.error(f"Error calculando retornos de estrategia: {e}")
            return pd.Series()
    
    def calculate_rsi(self, prices: pd.Series, period: int = 14) -> pd.Series:
        """Calcular RSI para backtesting"""
        try:
            delta = prices.diff()
            gain = (delta.where(delta > 0, 0)).rolling(window=period).mean()
            loss = (-delta.where(delta < 0, 0)).rolling(window=period).mean()
            rs = gain / loss
            return 100 - (100 / (1 + rs))
        except:
            return pd.Series(50, index=prices.index)
    
    def create_strategy_instance(self):
        """Crear instancia de la estrategia para backtesting"""
        # Simplificado para backtesting
        return type('Strategy', (), {
            'train': lambda self, data: None,
            'generate_signals': lambda self, data: pd.Series(0, index=data.index)
        })()

    def train_model(self, symbol: str) -> float:
        """Entrenar modelo con ensemble multi-temporal"""
        try:
            self.logger.info(f"Entrenando modelo multi-temporal para {symbol}...")
            
            # Preparar características
            features = self.prepare_features(symbol)
            if features.empty or len(features) < 500:
                self.logger.error("No se pudieron preparar suficientes características")
                return 0.0
            
            # Preparar etiquetas
            labels = self.prepare_labels(features)
            
            # Validación cruzada temporal
            if self.config.get('validation', {}).get('enabled', False):
                validation_results = self.temporal_validator.walk_forward_validation(
                    pd.concat([features, labels.rename('target')], axis=1),
                    MultiTemporalEnsemble,
                    {'config': self.config}
                )
                
                if validation_results:
                    accuracy = validation_results['aggregated_metrics']['accuracy']
                    self.logger.info(f"Validación walk-forward para {symbol}: accuracy={accuracy:.4f}")
            
            # Entrenar ensemble multi-temporal
            self.multi_temporal_ensemble.train_temporal_models(features, labels)
            
            # Entrenar modelo jerárquico
            self.hierarchical_model.train_level1_models(features, labels)
            self.hierarchical_model.train_meta_model(features, labels)
            
            # Obtener condiciones de mercado para entrenamiento de decisión
            market_conditions_list = []
            for i in range(len(features)):
                if i < 20:
                    market_conditions_list.append({})
                else:
                    market_conditions_list.append(self.analyze_market_conditions(features.iloc[:i+1]))
            
            market_conditions_df = pd.DataFrame(market_conditions_list)
            self.hierarchical_model.train_decision_model(features, labels, market_conditions_df)
            
            # Guardar modelo
            self.multi_temporal_ensemble.save_model(f'{symbol}_multi_temporal')
            self.hierarchical_model.save_model(f'{symbol}_hierarchical')
            
            # Registrar actualización
            self.last_model_update = datetime.now()
            self.model_update_history.append({
                'timestamp': self.last_model_update,
                'symbol': symbol,
                'model_type': 'multi_temporal_ensemble',
                'features_count': len(features.columns)
            })
            
            self.logger.info(f"Modelo multi-temporal entrenado para {symbol}")
            return accuracy if 'accuracy' in locals() else 0.8
            
        except Exception as e:
            self.logger.error(f"Error entrenando modelo para {symbol}: {e}")
            return 0.0

            for symbol in self.config['symbols']:
                try:
                    self.train_model(symbol)
                except Exception as e:
                    self.logger.error(f"Error entrenando modelo para {symbol}: {e}")
            
            # Bucle principal de trading
            while self.is_running:
                try:
                    # Verificar si se debe reentrenar por drift
                    if self.drift_detector.should_retrain():
                        self.logger.info("Detectada necesidad de reentrenamiento por drift")
                        for symbol in self.config['symbols']:
                            try:
                                self.train_model(symbol)
                            except Exception as e:
                                self.logger.error(f"Error reentrenando modelo para {symbol}: {e}")
                    
                    # Actualizar modelos periódicamente
                    if self.should_update_models():
                        for symbol in self.config['symbols']:
                            try:
                                self.train_model(symbol)
                            except Exception as e:
                                self.logger.error(f"Error actualizando modelo para {symbol}: {e}")
                    
                    # Procesar cada símbolo
                    for symbol in self.config['symbols']:
                        try:
                            # Obtener precio actual
                            ticker = self.exchange.fetch_ticker(symbol)
                            current_price = ticker['last']
                            
                            # Generar señal
                            signal, signal_strength, context = self.generate_signal(symbol)
                            
                            # Ejecutar operación si hay señal
                            if signal != 'hold':
                                self.execute_trade(symbol, signal, signal_strength, context)
                            
                            # Verificar posiciones abiertas
                            self.check_positions(symbol, current_price)
                            
                            # Actualizar estado del portafolio
                            portfolio_state = self.update_portfolio_state()
                            
                            # Detectar drift en predicciones
                            if hasattr(self, 'last_predictions') and hasattr(self, 'last_actuals'):
                                drift_detected = self.drift_detector.detect_drift(
                                    self.last_predictions, self.last_actuals
                                )
                                if drift_detected:
                                    self.logger.warning(f"Drift detectado para {symbol}")
                            
                            self.logger.info(
                                f"{symbol}: {signal} (Fuerza: {signal_strength:.2f}), "
                                f"Drawdown: {self.drawdown_level}, "
                                f"Portfolio: ${portfolio_state.get('total_value', 0):.2f}, "
                                f"PnL Diario: {portfolio_state.get('daily_pnl', 0):.2%}"
                            )
                            
                        except Exception as e:
                            self.logger.error(f"Error procesando {symbol}: {e}")
                    
                    # Esperar antes del siguiente ciclo
                    time.sleep(self.config['trading']['interval'])
                
                except KeyboardInterrupt:
                    self.logger.info("Bot detenido por el usuario")
                    self.is_running = False
                    break
                except Exception as e:
                    self.logger.error(f"Error en bucle principal: {e}")
                    time.sleep(60)
            
        except Exception as e:
            self.logger.error(f"Error fatal en ejecución: {e}")
    
    def stop(self):
        """Detener el bot de trading"""
        self.is_running = False
        self.logger.info("Bot de trading detenido")
