import yaml
import logging
from datetime import datetime
from quantum_trading_bot_pro import QuantumTradingBotPro

def main():
    # Configurar logging
    logging.basicConfig(
        level=logging.INFO,
        format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
        handlers=[
            logging.FileHandler('logs/quantum_fusion_pro.log'),
            logging.StreamHandler()
        ]
    )
    
    logger = logging.getLogger(__name__)
    logger.info("Iniciando Quantum Fusion Pro")
    
    try:
        # Crear y ejecutar bot
        bot = QuantumTradingBotPro('config.yaml')
        bot.run()
    except KeyboardInterrupt:
        logger.info("Bot detenido por el usuario")
    except Exception as e:
        logger.error(f"Error fatal: {e}")

if __name__ == "__main__":
    main()
