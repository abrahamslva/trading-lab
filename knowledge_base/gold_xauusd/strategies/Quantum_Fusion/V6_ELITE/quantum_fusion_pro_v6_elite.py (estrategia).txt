    def build_lstm_transformer_hybrid(self, input_size: int, sequence_length: int = 60):
        """Hybrid LSTM-Transformer para predicción de precios"""
        
        class LSTMTransformerHybrid(nn.Module):
            def __init__(self, input_size, hidden_size=256, num_layers=3, 
                        num_heads=8, dropout=0.1):
                super(LSTMTransformerHybrid, self).__init__()
                
                # LSTM layers para capturar dependencias temporales
                self.lstm = nn.LSTM(input_size, hidden_size, num_layers, 
                                  batch_first=True, dropout=dropout, bidirectional=True)
                
                # Transformer layers para attention patterns
                encoder_layer = nn.TransformerEncoderLayer(
                    d_model=hidden_size*2, nhead=num_heads, dim_feedforward=hidden_size*4,
                    dropout=dropout, activation='gelu'
                )
                self.transformer = nn.TransformerEncoder(encoder_layer, num_layers=4)
                
                # Attention mechanism
                self.attention = nn.MultiheadAttention(hidden_size*2, num_heads, dropout=dropout)
                
                # Output layers con múltiples objetivos
                self.price_head = nn.Sequential(
                    nn.Linear(hidden_size*2, hidden_size),
                    nn.GELU(),
                    nn.Dropout(dropout),
                    nn.Linear(hidden_size, 64),
                    nn.GELU(),
                    nn.Linear(64, 3)  # [direction, magnitude, confidence]
                )
                
                self.volatility_head = nn.Sequential(
                    nn.Linear(hidden_size*2, 64),
                    nn.GELU(),
                    nn.Linear(64, 1)  # volatility prediction
                )
                
                self.regime_head = nn.Sequential(
                    nn.Linear(hidden_size*2, 32),
                    nn.GELU(),
                    nn.Linear(32, 5)  # 5 market regimes
                )
                
            def forward(self, x):
                # LSTM processing
                lstm_out, (h_n, c_n) = self.lstm(x)
                
                # Transformer processing
                transformer_out = self.transformer(lstm_out.transpose(0, 1)).transpose(0, 1)
                
                # Self-attention
                attended, attention_weights = self.attention(
                    transformer_out.transpose(0, 1),
                    transformer_out.transpose(0, 1), 
                    transformer_out.transpose(0, 1)
                )
                
                # Use last sequence output
                final_output = attended[-1]
                
                # Multiple predictions
                price_pred = self.price_head(final_output)
                vol_pred = self.volatility_head(final_output)
                regime_pred = self.regime_head(final_output)
                
                return {
                    'price': torch.sigmoid(price_pred),
                    'volatility': torch.relu(vol_pred),
                    'regime': torch.softmax(regime_pred, dim=-1),
                    'attention_weights': attention_weights
                }
        
        return LSTMTransformerHybrid(input_size)
    
    def train_deep_models(self, X: pd.DataFrame, y: pd.DataFrame):

        vol_loss_fn = nn.MSELoss()
