            f[f'momentum_{scale}'] = returns.rolling(scale).sum()
            f[f'momentum_acceleration_{scale}'] = f[f'momentum_{scale}'].diff()
        
        # Fractal dimension (simplified Hurst exponent)
        def hurst_exponent(ts, max_lag=20):
            lags = range(2, max_lag)
            tau = [np.sqrt(np.std(np.subtract(ts[lag:], ts[:-lag]))) for lag in lags]
            poly = np.polyfit(np.log(lags), np.log(tau), 1)
            return poly[0] * 2.0
        
        f['hurst_exponent'] = pd.Series(index=close.index, dtype=float)
        for i in range(50, len(close)):
            window_returns = returns.iloc[i-50:i].dropna()
            if len(window_returns) > 20:
                f['hurst_exponent'].iloc[i] = hurst_exponent(window_returns.values)
        f['hurst_exponent'] = f['hurst_exponent'].fillna(0.5)
        
        # 4. CORRELATION DYNAMICS
        # Simplified cross-correlation with volume
        f['price_volume_correlation'] = pd.Series(index=close.index, dtype=float)
        for i in range(20, len(close)):
            price_window = returns.iloc[i-20:i]
            volume_window = volume.pct_change().iloc[i-20:i]
            if len(price_window) > 10 and len(volume_window) > 10:
                corr = np.corrcoef(price_window.dropna(), volume_window.dropna())[0, 1]
                f['price_volume_correlation'].iloc[i] = corr if not np.isnan(corr) else 0
        f['price_volume_correlation'] = f['price_volume_correlation'].fillna(0)
        
        # 5. REGIME CLASSIFICATION FEATURES
        # Volatility clustering
        vol = returns.rolling(20).std()
        f['volatility_clustering'] = vol / vol.rolling(50).mean()
        
        # Trend persistence
        trend_direction = np.sign(returns.rolling(10).sum())
        f['trend_persistence'] = (trend_direction == trend_direction.shift(1)).rolling(20).mean()
        
        # 6. MARKET STRUCTURE FEATURES
        # Support/Resistance strength
        for window in [20, 50, 100]:
            resistance = high.rolling(window).max()
            support = low.rolling(window).min()
            
            # Strength based on number of touches
            resistance_touches = (abs(high - resistance) < (high * 0.01)).rolling(window).sum()
            support_touches = (abs(low - support) < (low * 0.01)).rolling(window).sum()
            
            f[f'resistance_strength_{window}'] = resistance_touches
            f[f'support_strength_{window}'] = support_touches
            
            # Distance from key levels
            f[f'distance_from_resistance_{window}'] = (resistance - close) / close
            f[f'distance_from_support_{window}'] = (close - support) / close
        
        # 7. VOLUME MICROSTRUCTURE
        # Volume rate of change
        f['volume_roc'] = volume.pct_change()
        f['volume_acceleration'] = f['volume_roc'].diff()
        
        # Volume-Price Trend (VPT) with improvements
        vpt = (volume * returns).cumsum()
        f['vpt'] = vpt
        f['vpt_momentum'] = vpt.diff().rolling(10).mean()
        
        # Accumulation/Distribution with volume weighting
        ad_multiplier = ((close - low) - (high - close)) / (high - low + 1e-10)
        f['accumulation_distribution'] = (ad_multiplier * volume).cumsum()
        
        # 8. STATISTICAL FEATURES
        for window in [10, 20, 50]:
            # Higher moments
            f[f'returns_skew_{window}'] = returns.rolling(window).skew()
            f[f'returns_kurt_{window}'] = returns.rolling(window).kurt()
            
            # Percentile positions
            f[f'price_percentile_{window}'] = close.rolling(window*2).rank(pct=True)
            f[f'volume_percentile_{window}'] = volume.rolling(window*2).rank(pct=True)
            
            # Z-scores
            f[f'price_zscore_{window}'] = (close - close.rolling(window).mean()) / (close.rolling(window).std() + 1e-10)
            f[f'volume_zscore_{window}'] = (volume - volume.rolling(window).mean()) / (volume.rolling(window).std() + 1e-10)
        
        # 9. PATTERN RECOGNITION FEATURES
        # Candlestick patterns (enhanced)
        body_size = abs(close - open_price)
        upper_shadow = high - np.maximum(open_price, close)
        lower_shadow = np.minimum(open_price, close) - low
        total_range = high - low
        
        f['body_ratio'] = body_size / (total_range + 1e-10)
        f['upper_shadow_ratio'] = upper_shadow / (total_range + 1e-10)
        f['lower_shadow_ratio'] = lower_shadow / (total_range + 1e-10)
        
        # Gap analysis
        f['gap_up'] = (low > high.shift(1)).astype(int)
        f['gap_down'] = (high < low.shift(1)).astype(int)
        f['gap_size'] = np.where(f['gap_up'], low - high.shift(1), 
                                np.where(f['gap_down'], low.shift(1) - high, 0))
        
        # 10. TIME-BASED FEATURES (Enhanced)
        if hasattr(df.index, 'hour'):
            f['hour'] = df.index.hour
            f['day_of_week'] = df.index.dayofweek
            f['month'] = df.index.month
            
            # Session classification (more granular)
            f['asian_session'] = ((df.index.hour >= 23) | (df.index.hour <= 8)).astype(int)
            f['london_session'] = ((df.index.hour >= 7) & (df.index.hour <= 16)).astype(int)
            f['ny_session'] = ((df.index.hour >= 12) & (df.index.hour <= 21)).astype(int)
            f['overlap_session'] = ((df.index.hour >= 12) & (df.index.hour <= 16)).astype(int)
            
            # Weekend effect
            f['is_weekend'] = (df.index.dayofweek >= 5).astype(int)
            f['monday_effect'] = (df.index.dayofweek == 0).astype(int)
            f['friday_effect'] = (df.index.dayofweek == 4).astype(int)
        else:
            for col in ['hour', 'day_of_week', 'month', 'asian_session', 'london_session', 
                       'ny_session', 'overlap_session', 'is_weekend', 'monday_effect', 'friday_effect']:
                f[col] = 0
        
        # 11. INTERACTION FEATURES (Advanced)
        # Volatility-Volume interaction
        f['vol_volume_interaction'] = f['volatility_clustering'] * f['volume_zscore_20']
        
        # Momentum-Liquidity interaction
        f['momentum_liquidity'] = f['momentum_21'] * f['liquidity_asymmetry']
        
        # Trend-Volume confirmation
        f['trend_volume_confirmation'] = f['adaptive_trend_strength'] * f['volume_percentile_20']
        
        # Fill NaN values with sophisticated method
        f = f.fillna(method='ffill').fillna(method='bfill').fillna(0)
        
        # Feature scaling and clipping for stability
        for col in f.columns:
            if f[col].dtype in ['float64', 'int64']:
                # Clip extreme outliers
                q99 = f[col].quantile(0.99)
                q1 = f[col].quantile(0.01)
                f[col] = f[col].clip(q1, q99)
        
        return f

class DeepLearningEnsembleV6:
    def __init__(self, config: UltraConfigV6):
        self.config = config
        self.models = {}
        self.model_weights = {}
        self.feature_scalers = {}
        self.sequence_length = config.lstm_sequence_length
        
    def build_lstm_attention_model(self, input_shape: Tuple[int, int]) -> tf.keras.Model:
        """Build LSTM with Multi-Head Attention"""
        if not HAS_TENSORFLOW:
            return None
            
        model = Sequential([
            LSTM(self.config.lstm_hidden_units, return_sequences=True, input_shape=input_shape),
            Dropout(0.2),
            LSTM(self.config.lstm_hidden_units//2, return_sequences=True),
            Dropout(0.2),
            LSTM(self.config.lstm_hidden_units//4, return_sequences=False),
            Dropout(0.2),
            Dense(64, activation='relu'),
            Dropout(0.1),
            Dense(32, activation='relu'),
            Dense(1, activation='sigmoid')
        ])
        
        model.compile(
            optimizer=Adam(learning_rate=0.001),
            loss='binary_crossentropy',
            metrics=['accuracy', 'precision', 'recall']
        )
        
        return model
    
    def build_transformer_model(self, input_shape: Tuple[int, int]) -> tf.keras.Model:
        """Build Transformer-based model"""
        if not HAS_TENSORFLOW:
            return None
            
        inputs = tf.keras.Input(shape=input_shape)
        
        # Multi-head attention
        attention = MultiHeadAttention(
            num_heads=self.config.attention_heads,
            key_dim=input_shape[1]//self.config.attention_heads
        )(inputs, inputs)
        
        # Add & Norm
        attention = tf.keras.layers.Add()([inputs, attention])
        attention = tf.keras.layers.LayerNormalization()(attention)
        
        # Feed Forward
        ff = Dense(self.config.lstm_hidden_units, activation='relu')(attention)
        ff = Dropout(0.1)(ff)
        ff = Dense(input_shape[1])(ff)
        
        # Add & Norm
        ff = tf.keras.layers.Add()([attention, ff])
        ff = tf.keras.layers.LayerNormalization()(ff)
        
        # Global pooling and output
        pooled = tf.keras.layers.GlobalAveragePooling1D()(ff)
        outputs = Dense(1, activation='sigmoid')(pooled)
        
        model = tf.keras.Model(inputs=inputs, outputs=outputs)
        model.compile(
            optimizer=Adam(learning_rate=0.001),
            loss='binary_crossentropy',
            metrics=['accuracy']
        )
        
        return model
    
    def prepare_sequences(self, X: pd.DataFrame, y: pd.Series = None) -> Tuple[np.ndarray, np.ndarray]:
        """Prepare sequences for LSTM models"""
        # Scale features
        scaler = QuantileTransformer(n_quantiles=1000, output_distribution='normal')
        X_scaled = scaler.fit_transform(X)
        
        # Create sequences
        X_sequences = []
        y_sequences = [] if y is not None else None
        
        for i in range(self.sequence_length, len(X_scaled)):
            X_sequences.append(X_scaled[i-self.sequence_length:i])
            if y is not None:
                y_sequences.append(y.iloc[i])
        
        X_seq = np.array(X_sequences)
        y_seq = np.array(y_sequences) if y_sequences else None
        
        return X_seq, y_seq
    
    def train_deep_ensemble(self, X: pd.DataFrame, y: pd.Series):
        """Train deep learning ensemble"""
        if not HAS_TENSORFLOW:
            logging.warning("TensorFlow not available, skipping deep learning models")
            return
            
        # Prepare sequences
        X_seq, y_seq = self.prepare_sequences(X, y)
        
        if len(X_seq) < 100:  # Need sufficient data
            logging.warning("Insufficient data for deep learning training")
            return
        
        # Train/validation split
        split_idx = int(len(X_seq) * 0.8)
        X_train, X_val = X_seq[:split_idx], X_seq[split_idx:]
        y_train, y_val = y_seq[:split_idx], y_seq[split_idx:]
        
        # Callbacks
        early_stopping = EarlyStopping(patience=10, restore_best_weights=True)
        reduce_lr = ReduceLROnPlateau(factor=0.5, patience=5)
        
        # Train LSTM model
        try:
            lstm_model = self.build_lstm_attention_model((self.sequence_length, X.shape[1]))
            if lstm_model:
                lstm_model.fit(
                    X_train, y_train,
                    validation_data=(X_val, y_val),
                    epochs=100,
                    batch_size=32,
                    callbacks=[early_stopping, reduce_lr],
                    verbose=0
                )
                self.models['lstm_attention'] = lstm_model
                
                # Calculate model weight based on validation performance
                val_pred = lstm_model.predict(X_val)
                val_auc = roc_auc_score(y_val, val_pred)
                self.model_weights['lstm_attention'] = val_auc
                
        except Exception as e:
            logging.warning(f"LSTM training failed: {e}")
        
        # Train Transformer model
        try:
            transformer_model = self.build_transformer_model((self.sequence_length, X.shape[1]))
            if transformer_model:
                transformer_model.fit(
                    X_train, y_train,
                    validation_data=(X_val, y_val),
                    epochs=50,
                    batch_size=32,
                    callbacks=[early_stopping, reduce_lr],
                    verbose=0
                )
                self.models['transformer'] = transformer_model
                
                # Calculate model weight
                val_pred = transformer_model.predict(X_val)
                val_auc = roc_auc_score(y_val, val_pred)
                self.model_weights['transformer'] = val_auc
                
        except Exception as e:
            logging.warning(f"Transformer training failed: {e}")
        
        # Normalize weights
        total_weight = sum(self.model_weights.values())
        if total_weight > 0:
            self.model_weights = {k: v/total_weight for k, v in self.model_weights.items()}
    
    def predict_deep_ensemble(self, X: pd.DataFrame) -> Tuple[float, float]:
        """Predict using deep learning ensemble"""
        if not self.models:
            return 0.5, 0.0
        
        X_seq, _ = self.prepare_sequences(X)
        if len(X_seq) == 0:
            return 0.5, 0.0
        
        predictions = []
        weights = []
        
        for model_name, model in self.models.items():
            try:
                pred = model.predict(X_seq[-1:], verbose=0)[0][0]  # Latest sequence
                predictions.append(pred)
                weights.append(self.model_weights.get(model_name, 1.0))
            except Exception as e:
                logging.warning(f"Deep learning prediction failed for {model_name}: {e}")
        
        if not predictions:
            return 0.5, 0.0
        
        # Weighted average
        weighted_pred = np.average(predictions, weights=weights)
        
        # Confidence based on agreement
        confidence = 1.0 - np.std(predictions) if len(predictions) > 1 else 0.5
        
        return float(weighted_pred), float(confidence)
