## 🛠️ Código de Integración para Trading Lab

### FRED API — Series Clave para Gold Trading
```python
import requests, pandas as pd

FRED_KEY = "YOUR_KEY"  # Gratis en fred.stlouisfed.org
BASE = "https://api.stlouisfed.org/fred/series/observations"

SERIES = {
    "GOLDAMGBD228NLBM": "Gold London Fix USD/oz",
    "DTWEXBGS":         "DXY Trade-Weighted USD",
    "DGS10":            "US 10Y Treasury Yield",
    "VIXCLS":           "VIX Volatility",
    "CPIAUCSL":         "CPI Inflation YoY",
    "DFII10":           "Real Interest Rate 10Y (TIPS)",
    "T10YIE":           "Breakeven Inflation 10Y",
    "DEXUSEU":          "EUR/USD",
    "DEXJPUS":          "JPY/USD",
}

def get_fred(series_id, start="2020-01-01"):
    r = requests.get(BASE, params={
        "series_id": series_id,
        "api_key": FRED_KEY,
        "file_type": "json",
        "observation_start": start,
    })
    data = r.json()["observations"]
    df = pd.DataFrame(data)[["date","value"]].rename(columns={"value": series_id})
    df["date"] = pd.to_datetime(df["date"])
    df[series_id] = pd.to_numeric(df[series_id], errors="coerce")
    return df.set_index("date")
```

### yfinance — Sin API Key (Contexto Macro)
```python
import yfinance as yf

MACRO_TICKERS = {
    "GC=F":   "Gold Futures",
    "SI=F":   "Silver Futures",
    "^VIX":   "Volatility Index",
    "^TNX":   "US 10Y Yield",
    "UUP":    "DXY ETF proxy",
    "GLD":    "Gold ETF SPDR",
    "TLT":    "20Y Treasury ETF",
    "^GSPC":  "S&P 500",
    "CL=F":   "WTI Crude Oil",
    "BTC-USD":"Bitcoin (correlación)",
    "EURUSD=X":"EUR/USD",
    "JPYUSD=X":"JPY/USD",
}

def get_macro_yf(period="2y"):
    data = {}
    for ticker, name in MACRO_TICKERS.items():
        df = yf.download(ticker, period=period, interval="1d", progress=False)
        if not df.empty:
            data[ticker] = df["Close"]
    return pd.DataFrame(data)
```

def get_crypto_vs_gold(days=365):
    url = "https://api.coingecko.com/api/v3/coins/bitcoin/market_chart"
    r = requests.get(url, params={"vs_currency": "usd", "days": days})
    prices = r.json()["prices"]
    df = pd.DataFrame(prices, columns=["timestamp", "BTC_USD"])
